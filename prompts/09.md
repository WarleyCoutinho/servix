 async function updateUserRoleBasedOnPlan(barbershopId: string, plan: SubscriptionPlan) {  
 const barbershop = await prisma.barbershop.findUnique({  
 where: { id: barbershopId },  
 select: { ownerId: true },  
 });

if (!barbershop?.ownerId) return;

const newRole = plan === SubscriptionPlan.BASIC  
 ? UserRole.owner_professional  
 : UserRole.owner;

await prisma.user.update({  
 where: { id: barbershop.ownerId },

 });
